# 📊 SBP API — Памятка по статистике

# 🌐 Базовый URL
# http://192.168.10.130:8510


# 📈 HTML СТАТИСТИКА
# ========================================
# За последние 24 часа
curl "http://192.168.10.130:8510/stats/html?hours=24"

# За последние 48 часов
curl "http://192.168.10.130:8510/stats/html?hours=48"

# За последние 7 дней (168 часов)
curl "http://192.168.10.130:8510/stats/html?hours=168"

# За конкретную дату
curl "http://192.168.10.130:8510/stats/html?start=2026-09-01"

# За период
curl "http://192.168.10.130:8510/stats/html?start=2026-09-01&end=2026-09-02"

# С фильтром по базе
curl "http://192.168.10.130:8510/stats/html?hours=24&base=copy_ursv"


# 📊 JSON СТАТИСТИКА
# ========================================
# За последние 24 часа
curl "http://192.168.10.130:8510/stats?hours=24"

# За последние 48 часов
curl "http://192.168.10.130:8510/stats?hours=48"

# За период с фильтром
curl "http://192.168.10.130:8510/stats?start=2026-09-01&base=copy_ursv"


# 🧪 ТЕСТОВЫЕ ЭНДПОИНТЫ
# ========================================
# Тест callback
curl "http://192.168.10.130:8510/test"

# Тест ошибки
curl "http://192.168.10.130:8510/test-error"

# Корневой эндпоинт
curl "http://192.168.10.130:8510/"


# ⚙️ ОСНОВНЫЕ ЭНДПОИНТЫ
# ========================================
# Создать payload
curl "http://192.168.10.130:8510/create_payload/copy_ursv"

# Получить payload
curl "http://192.168.10.130:8510/get_payload/copy_ursv"

# Остановить payload
curl "http://192.168.10.130:8510/stop_payload/copy_ursv"

# Callback (POST)
curl -X POST "http://192.168.10.130:8510/callback" \
  -H "Content-Type: application/json" \
  -d '[{"requestId":"1779","qrPayload":"https://example.com/qr/123"}]'

# Очистка БД
curl "http://192.168.10.130:8510/clear_db/1000"

# Очистка одного запроса
curl "http://192.168.10.130:8510/clear_one_db/1779"

# VACUUM БД
curl "http://192.168.10.130:8510/vacuum"


# 📋 ПАРАМЕТРЫ
# ========================================
# hours  - За последние N часов (integer)
# start  - Начальная дата (YYYY-MM-DD)
# end    - Конечная дата (YYYY-MM-DD)
# base   - Фильтр по базе (string)


# 🎯 ГЛАВНАЯ СТАТИСТИКА (открыть в браузере)
# ========================================
# http://192.168.10.130:8510/stats/html?hours=24
